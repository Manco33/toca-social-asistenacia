💻 CÓDIGO PARA EJECUTAR:
═══════════════════════════════════════════════════════════════════════════════

EMPLEADOS (Agregar DPI y celular):

```sql
ALTER TABLE empleados
ADD COLUMN dpi VARCHAR(255) UNIQUE,
ADD COLUMN numero_celular VARCHAR(20);

CREATE INDEX idx_empleados_dpi ON empleados(dpi);
```

PROYECTOS (Cambiar ID a VARCHAR):

```sql
-- Opción 1: Crear tabla nueva (Recomendada)
CREATE TABLE proyectos_backup AS SELECT * FROM proyectos;
DROP TABLE proyectos;

CREATE TABLE proyectos (
  id VARCHAR(50) PRIMARY KEY,
  nombre VARCHAR(255) NOT NULL UNIQUE,
  ubicacion_del_proyecto VARCHAR(255),
  soluciones VARCHAR(255),
  horas_totales DECIMAL(10,2),
  horas_usadas_manual DECIMAL(10,2) DEFAULT 0,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX idx_nombre (nombre)
);

INSERT INTO proyectos SELECT CAST(id AS CHAR), nombre, ubicacion_del_proyecto, soluciones, horas_totales, horas_usadas_manual, created_at, updated_at FROM proyectos_backup;
DROP TABLE proyectos_backup;
```
